PROCEDURE "STONEMEN_USPGETCRFSEARCHDATA" (
    IN GUID NVARCHAR(50),
    IN BUYERCODE NVARCHAR(50),
    IN CRFREQNO INTEGER,
    IN CRFSTATUS NVARCHAR(100),
    IN CRFREQDATE NVARCHAR(15),
    IN Techno_Id NVARCHAR(200),
    OUT 
  ReturnData TABLE (
      CRFREQGUID NVARCHAR(255),
      CRFREQNO INTEGER,
      INPUTTYPE NVARCHAR(255),
      BUYERNAME NVARCHAR(255),
      USERNAME NVARCHAR(255),
      CRFSTATUS NVARCHAR(255),
      REQTYPE nvarchar(200)
    )
  ) 
  LANGUAGE SQLSCRIPT 
  SQL SECURITY INVOKER 
  READS SQL DATA AS 
BEGIN
  
    A1 =(
  SELECT DISTINCT A."CRFREQGUID",
    A."CRFREQNO",
    A."INPUTTYPE",
    A."BUYERNAME",
    'Ankita' AS "USERNAME",
    A."CRFSTATUS",
    A."REQTYP"
  FROM "STONEMEN_DCRFHEADER" AS A
  ORDER BY A."CRFREQNO" DESC
);

Returndata =
SELECT DISTINCT A."CRFREQGUID",
  A."CRFREQNO",
  CASE
    WHEN A."INPUTTYPE" = 'I' THEN 'Internal'
    WHEN A."INPUTTYPE" = 'E' THEN 'External'
    ELSE A."INPUTTYPE"
  END AS "INPUTTYPE",
  A."BUYERNAME",
  A."USERNAME",
  CASE
    WHEN "CRFSTATUS" = 'New' THEN 'New'
    WHEN "CRFSTATUS" = 'CLS' THEN '
Closed'
    WHEN "CRFSTATUS" = 'WIP' THEN 'In Progress'
  END "CRFSTATUS",
  CASE
    WHEN A."REQTYP" = 'N' THEN 'New'
    WHEN A."REQTYP" = 'R' THEN 'Repeat'
    ELSE A."REQTYP"
  END AS "REQTYPE"
FROM :A1 AS A
ORDER BY A."CRFREQNO" DESC;
END;
