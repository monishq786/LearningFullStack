
PROCEDURE COP_SEARCH
(
	IN LOGINUSERID        NVARCHAR(200),
	IN CADNO			  INTEGER,
	IN ITEMCODE 	   	  NVARCHAR(200),
    IN SCIPLCODE		  NVARCHAR(200),
    IN ITEMDES  	      NVARCHAR(200),
    IN BUYERNAME	      NVARCHAR(200),
    IN COSTINGCATEGORY    NVARCHAR(200),
    IN PRODUCTSUBCATEGORY NVARCHAR(200),
    IN RAWMATERIAL		  NVARCHAR(200),
    IN CostingNo		  INTEGER,
    OUT TCOSTINGHEADER TABLE 
    (
    	COSTINGHEADERUUID    NVARCHAR(200),
    	COSTINGNO            INTEGER,
    	CADNO   		     INTEGER, 
    	ITEMCODE		     NVARCHAR(200),
    	SCIPLCODE		     NVARCHAR(200), 
    	ITEMDESC		     VARCHAR(200), 
        BUYERNAME		     NVARCHAR(200),
        COSTINGCATEGORY      NVARCHAR(200), 
        PRODUCTSUBCATEGORY   NVARCHAR(200),
        COSTINGTYPECODE      NVARCHAR(200),
        FORMSTATUS           NVARCHAR(200),
		CREATEDBY             NVARCHAR(200),                         
        RAWMATERIAL          NVARCHAR(200),
        CostingDate          DATE
    )
)
AS BEGIN
--==============DECLARE VARIABLE FOR ACTUAL VALUE GET FOR FILTER =========================
	DECLARE A_ITEMCODE 	   	     NVARCHAR(200);
    DECLARE A_SCIPLCODE		     NVARCHAR(200);
    DECLARE A_ITEMDES  	         NVARCHAR(200);
    DECLARE A_BUYERNAME	         NVARCHAR(200);
    DECLARE A_COSTINGCATEGORY    NVARCHAR(200);
    DECLARE A_PRODUCTSUBCATEGORY NVARCHAR(200);
    DECLARE A_RAWMATERIAL		 NVARCHAR(200);
    DECLARE GUID_COUNT     	    INTEGER;
    
    IF ITEMCODE = 'null' THEN  A_ITEMCODE := NULL; ELSE A_ITEMCODE :=ITEMCODE; END IF ;
    IF SCIPLCODE = 'null' THEN  A_SCIPLCODE := NULL; ELSE A_SCIPLCODE :=SCIPLCODE; END IF;
    IF ITEMDES = 'null' THEN  A_ITEMDES := NULL; ELSE A_ITEMDES =:ITEMDES; END IF;
    IF BUYERNAME  = 'null' THEN  A_BUYERNAME := NULL; ELSE A_BUYERNAME :=BUYERNAME; END IF;
    IF COSTINGCATEGORY  = 'null' THEN  A_COSTINGCATEGORY := NULL; ELSE A_COSTINGCATEGORY :=COSTINGCATEGORY; END IF;
    IF PRODUCTSUBCATEGORY  = 'null' THEN  A_PRODUCTSUBCATEGORY := NULL; ELSE A_PRODUCTSUBCATEGORY :=PRODUCTSUBCATEGORY; END IF; 
    IF RAWMATERIAL  = 'null' THEN  A_RAWMATERIAL := NULL; ELSE A_RAWMATERIAL :=RAWMATERIAL; END IF;
    
    MARGEActual_TABLE=(
    	SELECT "COSTINGHEADERID_COSTINGHEADERUUID","ITEMCODE" FROM "SAP_CAPIRE_STONEMAN_TCOSTINGSTONEACTUALCOSTDETAIL" 
    	UNION ALL
    	SELECT "COSTINGHEADERID_COSTINGHEADERUUID","ITEMCODE" FROM "SAP_CAPIRE_STONEMAN_TCOSTINGWOODACTUALCOSTDETAIL"
    	union ALL
    	SELECT "COSTINGHEADERID_COSTINGHEADERUUID","ITEMCODE" FROM "SAP_CAPIRE_STONEMAN_TCOSTINGMETALACTUALCOSTDETAIL"
    );
    
	A=(	SELECT DISTINCT T1."COSTINGHEADERUUID", 	
		T1."CADNO",     T1."ITEMCODE",
		T1."SCIPLCODE", T1."ITEMDESC",
		T1."BUYERNAME", T1."COSTINGCATEGORY", T1."PRODUCTSUBCATEGORY",
		T1."COSTINGNO", T1."COSTINGTYPECODE", T1."FORMSTATUS",
		T2."USERID",    
		T2."USERNAME",
		T1.CostingDate,   
		T3."ITEMCODE" AS RAWMATERIAL
		FROM "SAP_CAPIRE_STONEMAN_TCOSTINGHEADER" AS T1  
		LEFT JOIN "SAP_CAPIRE_STONEMAN_MUSER" AS T2 ON T1."CREATEDBYUSERID_USERID" = T2."USERID"
		LEFT JOIN :MARGEActual_TABLE T3 ON T3.COSTINGHEADERID_COSTINGHEADERUUID =  T1."COSTINGHEADERUUID"
		WHERE T1."COSTING_EXECUTIVE_USERID" =:LOGINUSERID OR T1."COSTING_ATL_USERID" =:LOGINUSERID OR T1."COSTING_HEAD_USERID" =:LOGINUSERID
	);
	TCOSTINGHEADER=(
			SELECT DISTINCT
    			A."COSTINGHEADERUUID",    
    			A."COSTINGNO", 
    			A."CADNO",
    			A."ITEMCODE",         
    			A."SCIPLCODE", 
    			A."ITEMDESC",         
    			A."BUYERNAME", 
    			A."COSTINGCATEGORY",   
    			A."PRODUCTSUBCATEGORY",
				A."COSTINGTYPECODE", 
				A."FORMSTATUS",           
				A."USERNAME" as CREATEDBY,              
				A."RAWMATERIAL" AS RAWMATERIAL,
				A.CostingDate
				FROM :A AS A 
				WHERE (A."CADNO" = :CADNO OR (IFNULL(:CADNO, 0) = 0))
				AND (A."ITEMCODE" = :A_ITEMCODE OR (IFNULL(:A_ITEMCODE, '') = ''))
				AND (A."SCIPLCODE" = :A_SCIPLCODE OR (IFNULL(:A_SCIPLCODE, '') = '')) 
				AND (A."ITEMDESC" = :A_ITEMDES OR (IFNULL(:A_ITEMDES, '') = ''))
				AND (A."COSTINGCATEGORY" = :A_COSTINGCATEGORY OR (IFNULL(:A_COSTINGCATEGORY, '') = ''))
				AND (A."PRODUCTSUBCATEGORY" = :A_PRODUCTSUBCATEGORY OR (IFNULL(:A_PRODUCTSUBCATEGORY, '') = ''))    
				AND (A."RAWMATERIAL" = :A_RAWMATERIAL OR (IFNULL(:A_RAWMATERIAL, '') = ''))
				AND (A."BUYERNAME" = :A_BUYERNAME OR (IFNULL(:A_BUYERNAME, '') = ''))
				AND (A."COSTINGNO" = :CostingNo OR (IFNULL(:CostingNo, 0) = 0)) 
				ORDER BY A."COSTINGNO" DESC);
    
END;
