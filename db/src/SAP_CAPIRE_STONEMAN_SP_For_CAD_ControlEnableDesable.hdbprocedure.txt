Procedure SAP_CAPIRE_STONEMAN_SP_For_CAD_ControlEnableDesable(
	IN CADGUID varchar(200),
	IN USERGUID nvarchar(200),
	IN ROLECODE nvarchar(200),
	IN STAGECODE nvarchar(200),
	OUT EnableTable TABLE(
		ControlName nvarchar(200),
		ControlId nvarchar(200),
		Enabled boolean
	)
	
)

LANGUAGE SQLSCRIPT
SQL SECURITY INVOKER
READS SQL DATA
AS
BEGIN

	DECLARE COUNTOFCLOSEFROM INT;
	DECLARE SAVEORSUBMIT INT;
	DECLARE TRANSACTIONOPENORCLOSE INT ;
	
	
	SELECT COUNT(*) INTO COUNTOFCLOSEFROM FROM "SAP_CAPIRE_STONEMAN_TCADDETAIL" WHERE "CRFSTATUS" IN ('C','CLS') AND "CADDETAILUUID" =:CADGUID;
	SELECT COUNT(*) INTO SAVEORSUBMIT FROM "SAP_CAPIRE_STONEMAN_TCADDETAIL" WHERE "CADDETAILUUID" =:CADGUID AND "SAVEORSUBMIT" = 'SAVE' and DTPASSITANT_USERID = :USERGUID ;
	SELECT COUNT(*) INTO TRANSACTIONOPENORCLOSE FROM "SAP_CAPIRE_STONEMAN_TCADDETAILAPPROVALTRANSACTION" 
	WHERE "USERROLECODE_ROLECODE_ROLECONSTANT" =:ROLECODE 
	AND "STAGECODE_STAGECODE_STAGECONSTANT" =:STAGECODE
	AND "USERID_USERID" =:USERGUID;
	
	
	IF(:COUNTOFCLOSEFROM >0 )THEN
	
	EnableTable = SELECT T2."CONTROLNAME",T2."CONTROLID" ,CASE WHEN T2."ENABLED" = TRUE THEN FALSE END AS ENABLED  FROM "SAP_CAPIRE_STONEMAN_MCRFCONTROLS" AS T1 
		INNER JOIN "SAP_CAPIRE_STONEMAN_MCRFCONTROLDETAIL" AS T2 ON T1."CONTROLSID"= T2."PARENT_CONTROLSID" 
		WHERE T1."FORMTYPE" = 'CAD' ;
	
	ELSEIF(:TRANSACTIONOPENORCLOSE > 0) then 
	
		EnableTable = 	SELECT T2."CONTROLNAME",T2."CONTROLID" ,T2."ENABLED" FROM "SAP_CAPIRE_STONEMAN_MCRFCONTROLS" AS T1 
		INNER JOIN "SAP_CAPIRE_STONEMAN_MCRFCONTROLDETAIL" AS T2 ON T1."CONTROLSID"= T2."PARENT_CONTROLSID" 
		WHERE T1."FORMTYPE" = 'CAD'  AND T1."ROLECODE_ROLECODE_ROLECONSTANT" =:ROLECODE AND T1."STAGECODE_STAGECODE_STAGECONSTANT" =:STAGECODE;
	
	ELSEIF(:SAVEORSUBMIT > 0) THEN 
	
		EnableTable = 	SELECT T2."CONTROLNAME",T2."CONTROLID" ,T2."ENABLED" FROM "SAP_CAPIRE_STONEMAN_MCRFCONTROLS" AS T1 
		INNER JOIN "SAP_CAPIRE_STONEMAN_MCRFCONTROLDETAIL" AS T2 ON T1."CONTROLSID"= T2."PARENT_CONTROLSID" 
		WHERE T1."FORMTYPE" = 'CAD'  AND T1."ROLECODE_ROLECODE_ROLECONSTANT" =:ROLECODE AND T1."STAGECODE_STAGECODE_STAGECONSTANT" =:STAGECODE;
	
	else 
	
		EnableTable = 	SELECT DISTINCT T2."CONTROLNAME",T2."CONTROLID",CASE WHEN T2."ENABLED" = TRUE THEN FALSE ELSE COALESCE(T2."ENABLED", FALSE)END AS ENABLED
			FROM "SAP_CAPIRE_STONEMAN_MCRFCONTROLS" AS T1 INNER JOIN "SAP_CAPIRE_STONEMAN_MCRFCONTROLDETAIL" AS T2 ON T1."CONTROLSID" = T2."PARENT_CONTROLSID"
			WHERE T1."FORMTYPE" = 'CAD' AND T2."ENABLED" = TRUE;
		
	END IF;
	
	
   
   
END;
