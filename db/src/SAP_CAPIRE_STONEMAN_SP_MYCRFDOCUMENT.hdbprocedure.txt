PROCEDURE "SAP_CAPIRE_STONEMAN_SP_MYCRFDOCUMENT" ( 
    IN GUID NVARCHAR(50),
    IN BUYERCODE NVARCHAR(50),
    IN CRFREQNO INTEGER,
    IN CRFSTATUS NVARCHAR(100),
    IN CRFREQDATE NVARCHAR(15),
    IN Techno_Id NVARCHAR(200),
    OUT ReturnData TABLE
    (CRFREQUUID NVARCHAR(255),OLDCRFREQNOUUID NVARCHAR(255),OLDCRFREQNO NVARCHAR(255), CRFREQNO INTEGER,
    PROFLOW NVARCHAR(255),INPUTTYPE NVARCHAR(255),MERREQDATE DATE, BUYERCODE NVARCHAR(255),BUYERNAME NVARCHAR(255),
    CATEGORY NVARCHAR(255),ITEMCODE NVARCHAR(255),ITEMDESC NVARCHAR(255),ITEMGROUP NVARCHAR(255),ITEMGROUPNAME NVARCHAR(255),CRFDELDATE DATE,
    CARNO NVARCHAR(255),CRFREQDATE DATE,PDDATE DATE,PDNO NVARCHAR(255),USERNAME NVARCHAR(255),CRFSTATUS NVARCHAR(255),TECHNOUSERID VARCHAR(200),REQTYPE nvarchar(200))
)
LANGUAGE SQLSCRIPT
SQL SECURITY INVOKER
READS SQL DATA
AS
BEGIN
    DECLARE ActualBuyerCode NVARCHAR(200);
    DECLARE ActualCRFSTATUS NVARCHAR(200);
    DECLARE ActualCRFREQDATE NVARCHAR(200);
    DECLARE ActualTechno_Id NVARCHAR(200);
    DECLARE CountSeekAdvice NVARCHAR(15);
    -----handle null value for BuyerCode-------
    IF BUYERCODE = 'null' THEN
        ActualBuyerCode := NULL;
    ELSE
        ActualBuyerCode := BUYERCODE;
    END IF;
    -----handle null value for CrfStatus-------
    IF CRFSTATUS = 'null' THEN
        ActualCRFSTATUS := NULL;
    ELSE
        ActualCRFSTATUS := CRFSTATUS;
    END IF;
  -----handle null value for CrfReqDate-------
  IF CRFREQDATE = 'null' THEN
    ActualCRFREQDATE :=null; 
  ELSE
    ActualCRFREQDATE := CRFREQDATE;
  END IF;
    -----handle null value for Techno_id-------
    IF Techno_Id = 'null' THEN
      ActualTechno_Id := NULL; 
    ELSE
      ActualTechno_Id := Techno_Id;
    END IF;
    A1 =(
    SELECT DISTINCT A."CRFREQUUID", A."OLDCRFREQNOUUID", A."OLDCRFREQNO",A."CRFREQNO",
    A."PROFLOW",A."INPUTTYPE",A."MERREQDATE" ,A."BUYERCODE",  A."BUYERNAME",
    A."CATEGORY",A."ITEMCODE", A."ITEMDESC", A."ITEMGROUP" , A."ITEMGROUPNAME",A."CRFDELDATE", 
    A."CARNO",CASE WHEN "CRFREQDATE" IS NULL THEN CURRENT_DATE  ELSE "CRFREQDATE"  END AS CRFREQDATE,
    A."PDDATE", A."PDNO",C."USERNAME",A."CRFSTATUS",A."TECHNOUSERID_USERID",A."REQTYP"
    FROM "SAP_CAPIRE_STONEMAN_TCRFHEADER" AS A
    left JOIN "SAP_CAPIRE_STONEMAN_TCRFSEEKADVICE" AS B  ON A."CRFREQUUID" = B."CRFREQID_CRFREQUUID"
    left JOIN "SAP_CAPIRE_STONEMAN_MUSER" AS C  ON A."CREATEDBYUSERID_USERID" = C."USERID"
    left JOIN "SAP_CAPIRE_STONEMAN_TCRFINSPDRAWDETAIL" AS D ON A."CRFREQUUID" = D."CRFREQID_CRFREQUUID"
    WHERE (A."MERTEAMHEAD_USERID" = :GUID
    OR A."MERTL_USERID" = :GUID
    OR A."MERATL_USERID" = :GUID
    OR A."PDCUSERID_USERID" = :GUID
    OR A."TECHNOUSERID_USERID" = :GUID
    OR A."QUALITYTLUSERID_USERID" = :GUID
    OR A."QUALITYATLUSERID_USERID" = :GUID
    OR D."DRAFTUSERID_USERID"= :GUID
    OR B."USERID_USERID"=:GUID)
    ORDER BY A."CRFREQNO" DESC);
    
    Returndata = Select DISTINCT
    A."CRFREQUUID",
    A."OLDCRFREQNOUUID",
    A."OLDCRFREQNO",
    A."CRFREQNO",
    A."PROFLOW",
    CASE WHEN A."INPUTTYPE" = 'I' THEN 'Internal' 
    WHEN A."INPUTTYPE" =  'E' THEN 'External' ELSE A."INPUTTYPE" END AS "INPUTTYPE",
    -- A."INPUTTYPE",
    A."MERREQDATE" ,
    A."BUYERCODE", 
    A."BUYERNAME",
    A."CATEGORY",
    A."ITEMCODE",
    A."ITEMDESC",
    A."ITEMGROUP" ,
    A."ITEMGROUPNAME",
    A."CRFDELDATE", 
    A."CARNO",
    A."CRFREQDATE",
    A."PDDATE",
    A."PDNO",
    A."USERNAME",
    -- A."CRFSTATUS",
    case 
      when "CRFSTATUS" = 'New' then 'New'
      when "CRFSTATUS" = 'CLS' then 'Closed'
      when "CRFSTATUS" = 'WIP' then 'In Progress' end "CRFSTATUS",
    A."TECHNOUSERID_USERID" AS "TECHNOUSERID",
    CASE WHEN A."REQTYP" = 'N' THEN 'New' 
    WHEN A."REQTYP" =  'R' THEN 'Repeat' else A."REQTYP" end as "REQTYPE"
    -- A."REQTYP" as REQTYPE
    from :A1 AS A
     where (A."BUYERCODE" = :ActualBuyerCode OR (IFNULL(:ActualBuyerCode, '') = ''))
     AND (A."CRFREQNO" = :CRFREQNO OR (IFNULL(:CRFREQNO, 0) = 0))
     AND (A."CRFSTATUS" = :ActualCRFSTATUS OR (IFNULL(:ActualCRFSTATUS, '') = '')) 
     AND (A."TECHNOUSERID_USERID" = :ActualTechno_Id OR (IFNULL(:ActualTechno_Id, '') = '')) 
      AND ((:ActualCRFREQDATE IS NULL AND A."MERREQDATE" <= CURRENT_DATE)
      OR (:ActualCRFREQDATE IS NULL AND A."MERREQDATE" <= CURRENT_DATE)
      OR (A."MERREQDATE" = :ActualCRFREQDATE))  

    ORDER BY A."CRFREQNO" DESC;
END;
