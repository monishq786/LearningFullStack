PROCEDURE SAP_CAPIRE_STONEMAN_SP_FOR_CONTROLENABLEDISABLE( IN CRFGUID VARCHAR(200), IN LOGINGUID NVARCHAR(200), IN ROLECODE NVARCHAR(200), IN STAGECODE NVARCHAR(200), OUT EnableTable TABLE (
	ControlName NVARCHAR(200),
	ControlId NVARCHAR(200),
	Enabled BOOLEAN
) )
LANGUAGE SQLSCRIPT
SQL SECURITY INVOKER
READS SQL DATA AS
BEGIN
DECLARE FORMSTATUSCANCELORCLOSED NVARCHAR(100);
DECLARE INTCOUNT INT;

FORMSTATUSCANCELORCLOSED = '';

/* =============================================get Crf form status, if it is cancelled/closed =============================================*/
/*	SELECT 'C' INTO FORMSTATUSCANCELORCLOSED FROM "SAP_CAPIRE_STONEMAN_TCRFHEADER" WHERE "CRFREQUUID" = :CRFGUID and "CRFSTATUS" in ('C','CLS');*/
	SELECT CASE 
		WHEN COUNT(*) > 0 THEN 'C'
		ELSE ''
	  END
INTO FORMSTATUSCANCELORCLOSED
FROM "SAP_CAPIRE_STONEMAN_TCRFHEADER"
WHERE "CRFREQUUID" = :CRFGUID
AND "CRFSTATUS" IN ('C', 'CLS');


/*=============================================set View mode if Crf form is Closed or cancelled===============================================*/
IF :FORMSTATUSCANCELORCLOSED = 'C' THEN
EnableTable = SELECT 
		T1."CONTROLNAME",
		T1."CONTROLID",
		T1."ENABLED"
	FROM 
		"SAP_CAPIRE_STONEMAN_MCRFCONTROLS" AS T0
		INNER JOIN
		"SAP_CAPIRE_STONEMAN_MCRFCONTROLDETAIL" AS T1
		ON T0."CONTROLSID" = T1."PARENT_CONTROLSID"
	WHERE T1."ENABLED" = FALSE;
	else 
	SELECT 1
INTO INTCOUNT
FROM SAP_CAPIRE_STONEMAN_TCRFAPPROVALTRANSACTION
WHERE "USERID_USERID" = :LOGINGUID
	AND "USERROLECODE_ROLECODE_ROLECONSTANT" = :ROLECODE
	AND "STAGECODE_STAGECODE_STAGECONSTANT" = :STAGECODE
	AND "ROWSTATUS" = 'OPEN'
	AND "APPROVALSTATUS" = 'PENDING'
ORDER BY "ROWNUMBER" DESC
LIMIT 1;
-- Adjust this column to the one you want to sort by (e.g., timestamp or ID)
IF ( :INTCOUNT = 1 ) THEN
/*================================================task pending so fetch role & stage wise records============================================*/
EnableTable = SELECT 
		T1."CONTROLNAME",
		T1."CONTROLID",
		T1."ENABLED"
	FROM 
		"SAP_CAPIRE_STONEMAN_MCRFCONTROLS" AS T0
		INNER JOIN
		"SAP_CAPIRE_STONEMAN_MCRFCONTROLDETAIL" AS T1
		ON T0."CONTROLSID" = T1."PARENT_CONTROLSID"
	WHERE T0."ROLECODE_ROLECODE_ROLECONSTANT" = :ROLECODE
		AND T0.STAGECODE_STAGECODE_STAGECONSTANT = :STAGECODE;
ELSE

/*=================================stage and role mismatch so check user in seek advice of this crf==============================================*/
INTCOUNT = 0;
SELECT 1
	INTO INTCOUNT
	FROM "SAP_CAPIRE_STONEMAN_TCRFSEEKADVICE"
	WHERE "CRFREQID_CRFREQUUID" = :CRFGUID
		AND ("USERID_USERID" = :LOGINGUID
			OR "QUESTIONFROM_USERID" = :LOGINGUID);
IF :INTCOUNT > 0 THEN
	/*============================================user found in seek advice, identifier -"seek advice"===============================================*/
	EnableTable = SELECT 
			T1."CONTROLNAME",
			T1."CONTROLID",
			T1."ENABLED"
		FROM 
			"SAP_CAPIRE_STONEMAN_MCRFCONTROLS" AS T0
			INNER JOIN
			"SAP_CAPIRE_STONEMAN_MCRFCONTROLDETAIL" AS T1
			ON T0."CONTROLSID" = T1."PARENT_CONTROLSID"
		WHERE T1."ENABLED" = FALSE;
ELSE
	--T0."IDENTIFIER" ='SEEKADVICE';

	/*=======================================user not found in seek advice, identifier - "disabled"==================================================*/
	EnableTable = SELECT 
			T1."CONTROLNAME",
			T1."CONTROLID",
			T1."ENABLED"
		FROM 
			"SAP_CAPIRE_STONEMAN_MCRFCONTROLS" AS T0
			INNER JOIN
			"SAP_CAPIRE_STONEMAN_MCRFCONTROLDETAIL" AS T1
			ON T0."CONTROLSID" = T1."PARENT_CONTROLSID"
		WHERE T1."ENABLED" = FALSE;
END IF;
END IF;

END IF;
--T0."IDENTIFIER" ='DISABLE';

/*=============================================get 1, if last row task in approval table is OPEN=================================================*/
/*	select	1 INTO INTCOUNT FROM SAP_CAPIRE_STONEMAN_TCRFAPPROVALTRANSACTION --"SAP_CAPIRE_STONEMAN_TCRFHEADER" 
WHERE "USERID_USERID" = :LOGINGUID
AND "USERROLECODE_ROLECODE_ROLECONSTANT" = :ROLECODE 
AND "STAGECODE_STAGECODE_STAGECONSTANT"  = :STAGECODE
AND "ROWSTATUS" ='OPEN'
AND "APPROVALSTATUS"='PENDING';
*/

END
